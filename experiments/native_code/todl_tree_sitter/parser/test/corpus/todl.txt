namespace demo.types {
  import demo.base.Identifier;

  internal primitive Identifier : string {
    description = "A stable identifier";
    regex = "^[a-z]+$";
  }

  taxonomy TaskType : represents Task, demo.other.Work uses SharedTypes {
    description = "Kinds of task";
    term User {
      label = "User Task";
    }
    Task Automated {
      label = "Automated Task";
    }
  }

  viewpoint Process : frames Task, Event

  concept Task : Activity {
    description = "An activity";
    label : string;
    assignee : Identifier?;
    relationship incoming -> SequenceFlow[];
    invariant "Task ids are unique.";
    invariant {
      description = "The task has a type.";
      predicate = this.type == service implies this.assignee == none;
    }
    annotate Icon { name = "task"; }
  }
}
